{
    public struct Priority
    {
        public int Value { get; set; }
        public Priority(int value) => Value = value;
    }

    public class Task
    {
        public string Notes { get; set; }
        public DateTime DateCreated { get; set; }
        public Priority TaskPriority { get; set; }
        public DateTime? DueDate { get; set; }
        public bool IsComplete { get; set; }
        public bool Overdue => DueDate.HasValue && !IsComplete && DueDate.Value < DateTime.Now;
        private string _description;

        public Task(string description)
        {
            _description = description;
            DateCreated = DateTime.Now;
            TaskPriority = new Priority(1);
        }

        public string GetDescription() => _description;
        public void SetDescription(string desc) => _description = desc;
        public void ToggleCompletion() => IsComplete = !IsComplete;
        public override string ToString() => _description;
    }
}

using System.Collections.Generic;
using System.Linq;

namespace ProjectManagerProto.Models
{
    public class TaskList
    {
        private List<Task> _tasks = new List<Task>();
        private string _name;

        public int TotalTasksCount => _tasks.Count;
        public int IncompleteTasksCount => _tasks.Count(t => !t.IsComplete);

        public TaskList(string name) => _name = name;

        public string GetName() => _name;
        public void SetName(string name) => _name = name;
        public void AddTask(Task task) => _tasks.Add(task);
        public void DeleteAllCompletedTasks() => _tasks.RemoveAll(t => t.IsComplete);
        public IReadOnlyList<Task> GetTasks() => _tasks.AsReadOnly();
        public override string ToString() => _name;
    }
}

using System.Collections.Generic;
using System.Linq;

namespace ProjectManagerProto.Models
{
    public class TaskCollection
    {
        private List<TaskList> _taskLists = new List<TaskList>();

        public int TotalTasksCount => _taskLists.Sum(l => l.TotalTasksCount);
        public int IncompleteTasksCount => _taskLists.Sum(l => l.IncompleteTasksCount);

        public void AddTaskList(TaskList list) => _taskLists.Add(list);
        public List<TaskList> GetTaskLists() => _taskLists;
        public void DeleteAllCompletedTaskLists() => _taskLists.RemoveAll(l => l.IncompleteTasksCount == 0);
        public override string ToString() => $"TaskCollection: {_taskLists.Count} lists";
    }
}

namespace ProjectManagerProto
{
    public static class MauiProgram
    {
        public static MauiApp CreateMauiApp()
        {
            var builder = MauiApp.CreateBuilder();
            builder
                .UseMauiApp<App>()
                .ConfigureFonts(fonts =>
                {
                    fonts.AddFont("OpenSans-Regular.ttf", "OpenSansRegular");
                    fonts.AddFont("OpenSans-Semibold.ttf", "OpenSansSemibold");
                })
                .ConfigureLifecycleEvents(events =>
                {
#if WINDOWS
                    events.AddWindows(windows =>
                    {
                        windows.OnWindowCreated(window =>
                        {
                            const int width = 1200;
                            const int height = 800;
                            const int x = 100;
                            const int y = 100;

                            var mauiWinUIWindow = (Microsoft.UI.Xaml.Window)window;
                            var hwnd = WinRT.Interop.WindowNative.GetWindowHandle(mauiWinUIWindow);
                            var windowId = Microsoft.UI.Win32Interop.GetWindowIdFromWindow(hwnd);
                            var appWindow = AppWindow.GetFromWindowId(windowId);

                            if (appWindow is not null)
                            {
                                appWindow.MoveAndResize(new RectInt32(x, y, width, height));
                            }
                        });
                    });
#endif
                });

GitHub Copilot: Here’s how you can add double-tap (double-click) gesture support in .NET MAUI for your CollectionView items, and handle it in your ViewModel.

---

### 1. XAML: Add Double Tap Gesture to CollectionView Item

Add a `Border.GestureRecognizers` section inside your `Border` in the `DataTemplate`:

```xml
<Border Stroke="#CCC" StrokeThickness="1" Margin="0,0,0,8" Padding="12">
    <Border.GestureRecognizers>
        <TapGestureRecognizer
            NumberOfTapsRequired="2"
            Command="{Binding Source={RelativeSource AncestorType={x:Type vm:ProjectsViewModel}}, Path=ProjectDoubleTappedCommand}"
            CommandParameter="{Binding .}" />
    </Border.GestureRecognizers>
    <Grid RowDefinitions="Auto,Auto,Auto,Auto">
        <Label Text="{Binding Name}" FontSize="16" FontAttributes="Bold"/>
        <Label Grid.Row="1" Text="{Binding TotalTasksCount, StringFormat='Tasks: {0}'}"/>
        <Label Grid.Row="2" Text="{Binding PercentComplete, StringFormat='Complete: {0:F0}%'}"/>
        <Label Grid.Row="3" Text="{Binding DateCreated, StringFormat='Created: {0:yyyy-MM-dd}'}"/>
    </Grid>
</Border>
```

**Key points:**
- `NumberOfTapsRequired="2"` for double-tap.
- `Command` is bound to a command in your ViewModel.
- `CommandParameter="{Binding .}"` passes the tapped project.

---

### 2. ViewModel: Add Command and Handler

Update your ViewModel to include an `ICommand` and the handler:

```csharp
using System.Collections.ObjectModel;
using System.Windows.Input;
using ProjectManagerProto.Models;
using Microsoft.Maui.Controls;

namespace ProjectManagerProto.ViewModels
{
    public class ProjectsViewModel
    {
        public ObservableCollection<Project> Projects { get; }
        public ICommand ProjectDoubleTappedCommand { get; }

        public ProjectsViewModel()
        {
            var sample = SampleData.Create();
            Projects = new ObservableCollection<Project>();
            foreach (var list in sample.GetTaskLists())
            {
                if (list is Project project)
                    Projects.Add(project);
            }

            ProjectDoubleTappedCommand = new Command<Project>(OnProjectDoubleClicked);
        }

        private void OnProjectDoubleClicked(Project project)
        {
            // Dummy window for Windows only
#if WINDOWS
            var window = new Microsoft.Maui.Controls.Window
            {
                Page = new ContentPage
                {
                    Content = new Label
                    {
                        Text = $"Project: {project?.Name ?? "Unknown"}",
                        VerticalOptions = LayoutOptions.Center,
                        HorizontalOptions = LayoutOptions.Center
                    }
                },
                Title = "Project Details"
            };
            Application.Current.OpenWindow(window);
#endif
        }
    }
}
```

GitHub Copilot: You are encountering issues because WinUI ContentDialog enforces that only one dialog can be open per XamlRoot (window) at a time. If any modal dialog (including DisplayPromptAsync, DisplayAlert, or another ContentDialog) is open in any window, trying to open another ContentDialog in the same window will throw or hang.

This is a WinUI platform limitation, not a MAUI one. Each window manages its own dialog stack, and dialogs are strictly modal per window. Attempting to open a ContentDialog while another is open in the same window will fail.

To avoid this:
- Never open two dialogs at once in the same window.
- Always await the completion of any dialog before opening another.
- If you need to queue dialogs, do so in code (e.g., only show the next after the previous closes).

Dialogs in different windows are independent, but each window can only have one open at a time.

How to get this converter to use 2 params? I need DueDate and IsComplete

GitHub Copilot: You cannot bind two properties directly to a single Label’s Text property in XAML.  
To use multiple values in a converter, use a MultiBinding with an IMultiValueConverter.

Example:

1. Create an IMultiValueConverter:

```csharp
public class DueDateAndCompleteToTextConverter : IMultiValueConverter
{
    public object Convert(object[] values, Type targetType, object parameter, CultureInfo culture)
    {
        var dueDate = values[0] as DateTime? ?? DateTime.MinValue;
        var isComplete = values[1] as bool? ?? false;
        // Your logic here
        return isComplete ? "Done" : dueDate.ToString("yyyy-MM-dd");
    }

    public object[] ConvertBack(object value, Type[] targetTypes, object parameter, CultureInfo culture)
        => throw new NotImplementedException();
}
```

2. Register the converter in your ResourceDictionary:

```xml
<converters:DueDateAndCompleteToTextConverter x:Key="DueDateAndCompleteToTextConverter" />
```

3. Use MultiBinding in your Label:

```xml
<Label Grid.Row="2">
    <Label.Text>
        <MultiBinding Converter="{StaticResource DueDateAndCompleteToTextConverter}">
            <Binding Path="DueDate"/>
            <Binding Path="IsComplete"/>
        </MultiBinding>
    </Label.Text>
</Label>
```

This allows you to use both DueDate and IsComplete in your converter.
